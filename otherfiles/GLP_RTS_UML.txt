package unit {

abstract class Unit {
    - unitName : String
    - faction : String
    - hp : int
    - maxHp : int
    - populationCost : int
    - ambroisieCost : int
    - faithCost : int
    - attackDamage : int
    - attackSpeed : float
    - movementSpeed : float
    - attackRange : int
    - visionRange : int 
    - hpRegenRate : float 
    - tierLevel : int 
    - target : MobileElement
    - isInCombat : boolean
    - destination : Block
    + getPercentHP() : int 
}

    - currentResourceLoad : int
    - ressourceType : String
    - maxCargoCapacity : int
    - currentDeposit : ResourceDeposit
}

class Infantry {
    - shieldValue : int
}

class Cavalry {
    - troughObstacles : boolean
    - chargeBonusDamage : int
}

    - blastRadius : float
    - attackRange : float
}

}

}
